function friendlyError(e,status){

const msg=(e&&e.message)||'Unknown error';

if(e&&e.name==='AbortError'){

return 'Server AI terlalu lama merespons (lebih dari 40 detik). Coba kirim lagi sebentar lagi.';

}

if(e instanceof TypeError||/failed to fetch|networkerror|load failed/i.test(msg)){

return 'Tidak bisa terhubung ke server AI.\n\n'+

'Kemungkinan penyebab:\n'+

'• Internet sedang putus\n'+

'• Cloudflare Worker belum di-deploy atau URL-nya salah\n'+

'• Worker belum mengizinkan CORS (balasan OPTIONS)\n'+

'• Worker crash karena ANTHROPIC_API_KEY belum diisi\n\n'+

'Detail teknis: '+msg;

}

if(status===401||status===403)return 'API key ditolak. Cek ANTHROPIC_API_KEY di Worker.\n\n'+msg;

if(status===429)return 'Terlalu banyak permintaan. Tunggu sebentar lalu coba lagi.\n\n'+msg;

return 'Maaf, AI belum bisa menjawab.\n\n'+msg;

}
